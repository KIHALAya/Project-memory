\usepackage[
  unicode,
  pdfencoding  = auto,
  colorlinks   = true,
  linkcolor    = LinkColor,
  citecolor    = CiteColor,
  urlcolor     = UrlColor,
  filecolor    = LinkColor,
  menucolor    = LinkColor,
  runcolor     = LinkColor,
  bookmarks    = true,
  bookmarksopen= true,
  bookmarksnumbered = true,
  pdfstartview = FitH,
  pdfdisplaydoctitle = true,
  pdfpagelayout= TwoPageRight,
  pdfpagemode  = UseOutlines,
  breaklinks   = true,
  hidelinks    = false
]{hyperref}

  \hypersetup{

    pdfauthor   = {\PDFAuthor},
    pdfsubject  = {\PDFSubject},
    pdfkeywords = {\PDFKeywords},
    pdfcreator  = {LuaLaTeX + biblatex + glossaries-extra},
    pdfproducer = {LaTeX with hyperref}
  }%
}
